// Package extproc_test — KAOS actor->resource authorization proof-of-concept.
//
// This file verifies (in-process, exercising the real ext_proc gRPC Process()
// flow + mock token exchange) that AIB #222's OPA hook, driven by an
// operator-supplied Rego policy, can enforce coarse agent(actor)->resource
// allow/deny for a KAOS-style multi-agent delegation topology.
//
// Scenario (CRD-grounded):
//
//	Agent A: agentNetwork.access:[B]  mcpServers:[X]
//	Agent B: agentNetwork.access:[C]  mcpServers:[Y]
//
// Matrix under test:
//
//	A -> X (mcp)         allow
//	A -> Y (mcp)         deny
//	B -> Y (mcp)         allow
//	B -> X (mcp)         deny
//	A -> B (delegation)  allow
//	A -> C (delegation)  deny
//	A -> X, no bearer    PASSTHROUGH  (documents the autonomous no-bearer gap)
//
// Focused run:
//
//	cd tests/e2e/extproc && go test -run TestExtprocE2E -args -ginkgo.focus="KAOS actor"
package extproc_test

import (
	"context"
	"encoding/base64"
	"encoding/json"

	extprocv3 "github.com/envoyproxy/go-control-plane/envoy/service/ext_proc/v3"
	. "github.com/onsi/ginkgo/v2"
	. "github.com/onsi/gomega"
	"google.golang.org/grpc"

	"github.com/agentic-identity-broker/agentic-identity-broker/tests/e2e/extproc/bootstrap"
	"github.com/agentic-identity-broker/agentic-identity-broker/tests/e2e/extproc/fixtures"
	"github.com/agentic-identity-broker/agentic-identity-broker/tests/e2e/extproc/helpers"
)

// actorToken builds an unsigned JWT (header.payload.signature) carrying the
// given agent identity as the `sub` claim. The signature segment is present but
// not verified — the Rego policy uses io.jwt.decode (no verification) since the
// axis under test is "can the policy read the actor identity", not crypto.
func actorToken(sub string) string {
	enc := func(v any) string {
		b, _ := json.Marshal(v)
		return base64.RawURLEncoding.EncodeToString(b)
	}
	header := enc(map[string]string{"alg": "none", "typ": "JWT"})
	payload := enc(map[string]string{"sub": sub, "azp": sub})
	sig := base64.RawURLEncoding.EncodeToString([]byte("unsigned"))
	return header + "." + payload + "." + sig
}

var _ = Describe("KAOS actor->resource authorization (AIB #222 OPA hook)", func() {
	var (
		env        *bootstrap.TestEnvironment
		grpcClient extprocv3.ExternalProcessorClient
		conn       *grpc.ClientConn
	)

	BeforeEach(func() {
		cfg := opaEnabledConfig(policyPath("kaos_actor.rego"))
		env = bootstrap.NewTestEnvironment(cfg, opaLogger)
		env.Start()

		env.MockTokenExchange.WithExchangedToken(fixtures.FreshExchangedToken)
		expiresIn := fixtures.StandardExpiresIn
		env.MockTokenExchange.WithExpiresIn(&expiresIn)

		grpcClient, conn = env.NewExtProcClient()
	})

	AfterEach(func() {
		conn.Close() //nolint:errcheck
		env.Stop()
	})

	// sendAs drives one body-bearing MCP request as `agent` targeting `host`,
	// returning the body-phase ProcessingResponse (the OPA decision point).
	sendAs := func(agent, host string) *extprocv3.ProcessingResponse {
		req := helpers.NewRequestHeaders().
			WithPath(fixtures.ValidResourceURI).
			WithBearerToken(fixtures.ValidBearerToken).
			WithHeader(":method", "POST").
			WithHeader(":authority", host).
			WithHeader("x-agent-authorization", actorToken(agent)).
			WithAgentgatewayProtocol("mcp").
			BuildWithMetadata()
		_, bodyResp := helpers.SendHeadersAndBody(
			context.Background(), grpcClient, req, toolCallBody("list_repositories"))
		return bodyResp
	}

	DescribeTable("enforces the declared delegation + MCP topology",
		func(agent, host string, allowed bool) {
			bodyResp := sendAs(agent, host)
			Expect(bodyResp).NotTo(BeNil())
			if allowed {
				Expect(bodyResp).To(helpers.BeForwardedRequestBody(),
					"%s -> %s should be ALLOWED by the actor->resource policy", agent, host)
			} else {
				Expect(bodyResp).To(helpers.HaveImmediateResponseWithStatus(403),
					"%s -> %s should be DENIED by the actor->resource policy", agent, host)
			}
		},
		Entry("A -> X (its MCP) allowed", "agent-A", "mcp-x:9003", true),
		Entry("A -> Y (B's MCP) denied", "agent-A", "mcp-y:9003", false),
		Entry("B -> Y (its MCP) allowed", "agent-B", "mcp-y:9003", true),
		Entry("B -> X (A's MCP) denied", "agent-B", "mcp-x:9003", false),
		Entry("A -> B (delegation) allowed", "agent-A", "agent-b:9003", true),
		Entry("A -> C (not permitted) denied", "agent-A", "agent-c:9003", false),
	)

	// GAP: autonomous / no-user-subject case. #222 gates OPA behind the presence
	// of an Authorization bearer — with no bearer it passes through WITHOUT
	// evaluating OPA. This asserts (and thereby documents) that the request is
	// NOT authorized: the headers phase is a plain pass-through (no exchange,
	// no OPA), even though the actor token is present and the policy would deny.
	It("does NOT enforce when there is no Authorization bearer (autonomous gap)", func() {
		req := helpers.NewRequestHeaders().
			WithPath(fixtures.ValidResourceURI).
			WithoutAuthorizationHeader().
			WithHeader(":method", "POST").
			WithHeader(":authority", "mcp-y:9003"). // A is NOT granted mcp-y
			WithHeader("x-agent-authorization", actorToken("agent-A")).
			WithAgentgatewayProtocol("mcp").
			BuildWithMetadata()

		headersResp, _ := helpers.SendHeadersAndBody(
			context.Background(), grpcClient, req, toolCallBody("list_repositories"))

		Expect(headersResp).NotTo(BeNil())
		Expect(headersResp).To(helpers.BePassThroughResponse(),
			"no-bearer request bypasses OPA entirely — the actor->resource policy is never consulted")
	})

	It("denies an unknown/unmapped actor (fail closed)", func() {
		bodyResp := sendAs("agent-Z", "mcp-x:9003")
		Expect(bodyResp).To(helpers.HaveImmediateResponseWithStatus(403),
			"an actor with no grants must be denied")
	})
})

// Same enforcement, but the grant graph is NOT in the rego — it is loaded from a
// sibling data document (data.kaos.grants), proving the permission graph is DATA
// (which KAOS's sync-service would project from CRDs and serve as an OPA bundle),
// not hardcoded policy. The rego rules are identical/static across deployments.

	var (
		env        *bootstrap.TestEnvironment
		grpcClient extprocv3.ExternalProcessorClient
		conn       *grpc.ClientConn
	)

	BeforeEach(func() {
		// Point #222 at the bundle DIRECTORY: it loads both the .rego rules and
		// data.json (data.kaos.grants) from the directory in path mode.
		cfg := opaEnabledConfig(policyPath("kaos_bundle"))
		env = bootstrap.NewTestEnvironment(cfg, opaLogger)
		env.Start()
		env.MockTokenExchange.WithExchangedToken(fixtures.FreshExchangedToken)
		expiresIn := fixtures.StandardExpiresIn
		env.MockTokenExchange.WithExpiresIn(&expiresIn)
		grpcClient, conn = env.NewExtProcClient()
	})

	AfterEach(func() {
		conn.Close() //nolint:errcheck
		env.Stop()
	})

	sendAs := func(agent, host string) *extprocv3.ProcessingResponse {
		req := helpers.NewRequestHeaders().
			WithPath(fixtures.ValidResourceURI).
			WithBearerToken(fixtures.ValidBearerToken).

			WithHeader("x-agent-authorization", actorToken(agent)).
